### Derived State (useMemo)

**1. Ownership-filtered tickets:**
```typescript
const ownershipTickets = useMemo(() => {
  if (!user) return [];
  switch (activeTab) {
    case 'my': return tickets.filter(t => t.submitted_by === user.id);
    case 'assigned': return tickets.filter(t => t.assigned_to === user.id);
    case 'all': return tickets; // admin only
    default: return [];
  }
}, [tickets, user, activeTab]);
```

**2. Fully filtered and sorted tickets:**
```typescript
const filteredTickets = useMemo(() => {
  return ownershipTickets
    .filter(t => {
      if (statusFilter !== 'All' && t.status !== statusFilter) return false;
      if (!categoryFilter.has(t.category)) return false;
      if (!urgencyFilter.has(t.urgency)) return false;
      if (searchTerm) {
        const term = searchTerm.toLowerCase();
        if (!t.subject.toLowerCase().includes(term) && !t.token.toLowerCase().includes(term)) return false;
      }
      return true;
    })
    .sort((a, b) => {
      // Sort by urgency (Critical→High→Medium→Low) then by created_at descending
      const urgencyA = URGENCY_SORT_ORDER.indexOf(a.urgency);
      const urgencyB = URGENCY_SORT_ORDER.indexOf(b.urgency);
      if (urgencyA !== urgencyB) return urgencyA - urgencyB;
      return new Date(b.created_at).getTime() - new Date(a.created_at).getTime();
    });
}, [ownershipTickets, statusFilter, categoryFilter, urgencyFilter, searchTerm]);
```

**3. KPI stats:**
```typescript
const stats = useMemo(() => ({

  resolved: filteredTickets.filter(t => t.status === 'Resolved').length,
}), [filteredTickets]);
```
