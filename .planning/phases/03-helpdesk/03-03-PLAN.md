### Submit Handler

```typescript
async function handleSubmit() {
  if (!category || !subject.trim() || !description.trim()) {
    setError('Please select a category and fill in all required fields.');
    return;
  }
  if (!user) { setError('You must be logged in to submit a ticket.'); return; }

  setSubmitting(true);
  setError(null);

  const result = await createTicket({
    subject: subject.trim(),
    category,
    urgency,
    description: description.trim(),
    submitted_by: user.id,
  });

  setSubmitting(false);

  if (result.success) {
    const ticketId = (result.data as { ticketId: string }).ticketId;
    setSuccess({ token: 'Ticket created successfully', ticketId });
    await refreshData?.();
  } else {
    setError(result.error ?? 'Failed to create ticket. Please try again. If the problem persists, contact your system administrator.');
  }
}
```
